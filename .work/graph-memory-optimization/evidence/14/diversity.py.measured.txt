"""Fresh-process high-definition-diversity public controls."""

import gc
import json
import resource
import subprocess
import sys
import time
import tracemalloc
from pathlib import Path

ROOT = Path(__file__).resolve().parents[4]
sys.path.insert(0, str(ROOT))


def sample(mode, traced):
    if traced:
        tracemalloc.start()
    from clean_ioc import ContainerBuilder

    started = time.perf_counter()
    builder = ContainerBuilder()
    services = [type(f"Service{index}", (), {}) for index in range(2000)]
    for service in services:
        builder.register(service)
    builder.set_provider_roots(())
    if mode == "rich-run":
        builder.add_validation_rule(
            lambda context: () if context.graph.diagnostics_enabled else (_ for _ in ()).throw(AssertionError())
        )
    runtime = builder.build(mode="analysis" if mode == "analysis" else "run")
    seconds = time.perf_counter() - started
    del builder
    gc.collect()
    memory = tracemalloc.get_traced_memory() if traced else (None, None)
    result = {
        "build_seconds": seconds,
        "traced_retained_bytes": memory[0],
        "traced_peak_bytes": memory[1],
        "peak_rss_bytes": resource.getrusage(resource.RUSAGE_SELF).ru_maxrss,
        "services": len(services),
    }
    started = time.perf_counter()
    for service in services:
        assert type(runtime.resolve(service)) is service
    result["resolve_seconds"] = time.perf_counter() - started
    runtime.__exit__(None, None, None)
    return result


if __name__ == "__main__":
    if sys.argv[1] == "sample":
        print(json.dumps(sample(sys.argv[2], sys.argv[3] == "heap")))
    else:
        output = {}
        for mode in ("run", "rich-run", "analysis"):
            output[mode] = []
            for repeat in range(3):
                for kind in ("normal", "heap"):
                    result = json.loads(
                        subprocess.check_output([sys.executable, __file__, "sample", mode, kind], text=True)
                    )
                    output[mode].append({"kind": kind, **result})
                    print(mode, repeat + 1, kind, result["build_seconds"], flush=True)
            Path(sys.argv[2]).write_text(json.dumps(output, indent=2) + "\n")
